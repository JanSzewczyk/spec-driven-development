import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Canonical phases" icon="list-format">
    `constitution → spec → clarify → plan → tasks → implement → review`, one slash command per phase.
  </Card>
  <Card title="Bring your own implementer" icon="puzzle">
    `/sdd:implement` hands each task to a skill, agent or command you configure, or implements it in the session.
  </Card>
  <Card title="Verified against the spec" icon="approve-check">
    `/sdd:review` runs tests, `spec-verifier` and your extra checks, then `reviewer` returns READY or NOT READY.
  </Card>
  <Card title="Any layout, any app" icon="seti:folder">
    Single app, monorepo or several repositories. Web, backend and mobile projects side by side, no fixed list of stacks.
  </Card>
</CardGrid>
